\centering
\caption{Nested Consumption Structure}
\label{tab:consumption_structure}
\begin{tabular}{clp{5cm}p{4cm}}
\toprule
\textbf{Level} & \textbf{Aggregator} & \textbf{Description} & \textbf{Parameters} \\
\midrule
\multirow{3}{*}{\textbf{1}} 
& \multirow{3}{*}{Stone-Geary} 
& Total consumption split between goods and services categories
& $\Omega_G = \bar{\omega} = 0.57$ \\
& & Fixed expenditure shares (non-homothetic preferences)
& $\Omega_S = 1-\bar{\omega} = 0.43$ \\
& & \multicolumn{2}{l}{$C = \Omega_G \cdot \frac{C_g}{P_g} + \Omega_S \cdot \frac{C_s}{P_s}$} \\
\midrule
\multirow{3}{*}{\textbf{2}} 
& \multirow{3}{*}{Cobb-Douglas} 
& Within-category allocation across 12 sectors
& $\gamma^g_i$ (from data) \\
& & Constant expenditure shares within each category
& $\gamma^s_i$ (from data) \\
& & \multicolumn{2}{l}{$C_g = \prod_{i \in \text{Goods}} C_{g,i}^{\gamma^g_i}$, \quad $C_s = \prod_{i \in \text{Services}} C_{s,i}^{\gamma^s_i}$} \\
\midrule
\multirow{4}{*}{\textbf{3}} 
& \multirow{4}{*}{Armington CES} 
& Home vs. foreign variety aggregation for each sector
& $\varrho_i$ (from data) \\
& & Imperfect substitution between domestic and imported goods
& $\sigma^H = 0.999$ \\
& & \multicolumn{2}{l}{$C_i = \left[\varrho_i^\sigma C_{H,i}^{(\sigma-1)/\sigma} + (1-\varrho_i)^\sigma C_{F,i}^{(\sigma-1)/\sigma}\right]^{\sigma/(\sigma-1)}$} \\
& & \multicolumn{2}{l}{$P_i = \left[\varrho_i^\sigma P_{H,i}^{1-\sigma} + (1-\varrho_i)^\sigma P_V^{1-\sigma}\right]^{1/(1-\sigma)}$} \\
\bottomrule
\end{tabular}
\begin{tablenotes}[flushleft]
\small
\item \textbf{Level 1:} Households allocate a fixed share of expenditure (57\%) to goods and (43\%) to services, independent of relative prices.
\item \textbf{Level 2:} Within each category, expenditure is allocated across sectors according to Cobb-Douglas weights $\gamma^g_i$ (for goods) and $\gamma^s_i$ (for services), constructed from Chilean consumption data. These sum to unity within each category.
\item \textbf{Level 3:} For each sector $i$, consumers choose between domestically-produced ($C_{H,i}$) and imported ($C_{F,i}$) varieties with home bias parameter $\varrho_i$ (typically close to 1, indicating strong preference for domestic goods) and elasticity of substitution $\sigma^H \approx 1$ (nearly Cobb-Douglas).
\item \textbf{Notation:} $P_{H,i}$ = domestic price, $P_V = Q \cdot P_V^*$ = import price in domestic currency, $Q$ = nominal exchange rate.
\end{tablenotes}
\end{table}
